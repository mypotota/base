# H16 风扇控制系统服务实作教程

在 AOSP 8.1 中增加 `h16_fan` Binder 系统服务。它可以查询 GPIO174 的使能状态、切回自动温控、临时打开或临时关闭风扇。下方源码按文件给齐，可逐个添加到 `~/aosp`。示例需要在你的 H16 上编译和验证。

实机已经确认 GPIO174 输出 `1` 会启动风扇，实际设备节点是 `/sys/devices/gpiochip0/gpio/gpio174/value`。以下代码的状态表示 GPIO 回读结果。要确认风扇确实在转，仍需观察气流或转速反馈。参见[风扇实测与自动温控修复](风扇实测与自动温控修复.md)。

```text
特权客户端 → h16_fan Binder → H16FanService（system_server）
                           → /dev/socket/h16_fand → h16_fand → GPIO174
```

## 1. 定义行为

- 启动进入 `AUTO`。高于或等于 75 °C 打开，低于或等于 60 °C 关闭，两者之间保持当前输出。示例取实机上 `thermal_zone0` 至 `thermal_zone6` 的最大有效读数，每秒更新，且有任何一个节点读失败时打开风扇。这是示例采用的温控策略，部署前请核对各温区、负载响应以及原厂温度选择逻辑。
- `ON`、`OFF` 是 10 秒测试模式。到期回到 AUTO。只在所有温区读数有效且最高温低于 60 °C 时接受 `OFF`。关闭期间达到 60 °C 或测温失败，立即结束手动模式并打开风扇。
- 响应是四个整数 `[gpio, mode, result, temperature]`。GPIO 为 `0`、`1` 或 `-1`（未知），模式为 `0=AUTO`、`1=ON`、`2=OFF`，结果为 `0=成功`、`1=保护拒绝`、`2=I/O 失败`，温度单位是毫摄氏度，`-1` 表示未知。查询使用 GPIO 实际回读。
- 现有 ramdisk 负责导出 GPIO174 并将其配置为输出。新进程负责运行期间唯一的写入。两个服务共用此 GPIO 时会互相覆盖，部署时应停用旧 `service fan` 的自动启动。

## 2. 原生进程

在 `device/skydroid/arowana_rc/h16_fan/` 新建以下两个文件。

### h16_fand.cpp

```cpp
#define LOG_TAG "h16_fand"

#include <algorithm>
#include <chrono>
#include <cerrno>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <fcntl.h>
#include <poll.h>
#include <sys/socket.h>
#include <sys/time.h>
#include <unistd.h>

#include <cutils/sockets.h>
#include <log/log.h>

namespace {

using Clock = std::chrono::steady_clock;

constexpr char kGpio[] = "/sys/devices/gpiochip0/gpio/gpio174/value";
constexpr int kHigh = 75000;
constexpr int kLow = 60000;
constexpr uid_t kSystemUid = 1000;

enum Mode { kAuto, kOn, kOff };
enum Result { kOk, kRejected, kIoError };

int readNumber(const char* path) {
    FILE* file = std::fopen(path, "r");
    if (file == nullptr) return -1;
    int value = -1;
    if (std::fscanf(file, "%d", &value) != 1) value = -1;
    std::fclose(file);
    return value;
}

int readGpio() {
    const int value = readNumber(kGpio);
    return value == 0 || value == 1 ? value : -1;
}

bool writeGpio(bool on) {
    const int fd = open(kGpio, O_WRONLY | O_CLOEXEC);
    if (fd < 0) return false;
    const char* value = on ? "1\n" : "0\n";
    const bool ok = write(fd, value, 2) == 2;
    close(fd);
    return ok && readGpio() == (on ? 1 : 0);
}

int readMaxTemperature() {
    int maximum = 0;
    for (int zone = 0; zone <= 6; ++zone) {
        char path[96];
        std::snprintf(path, sizeof(path),
                      "/sys/class/thermal/thermal_zone%d/temp", zone);
        const int value = readNumber(path);
        if (value < 0 || value > 150000) return -1;
        maximum = std::max(maximum, value);
    }
    return maximum;
}

class FanControl {
public:
    // 用单线程串行处理自动调节和客户端写入，避免 GPIO 竞争。
    Result update(char command) {
        temperature_ = readMaxTemperature();
        const auto now = Clock::now();
        const bool mustCancelOff = mode_ == kOff &&
                (temperature_ < 0 || temperature_ >= kLow);
        if (mode_ != kAuto && (now >= deadline_ || mustCancelOff)) {
            mode_ = kAuto;
        }

        Result result = kOk;
        switch (command) {
            case 'A': mode_ = kAuto; break;
            case '1': setTimedMode(kOn, now); break;
            case '0':
                if (temperature_ < 0 || temperature_ >= kLow) {
                    result = kRejected;
                } else {
                    setTimedMode(kOff, now);
                }
                break;
            case '?': break;
            default: return kRejected;
        }

        const int current = readGpio();
        bool target = current != 0;
        if (mode_ == kOn || temperature_ < 0 || temperature_ >= kHigh ||
                mustCancelOff) {
            target = true;
        } else if (mode_ == kOff || temperature_ <= kLow) {
            target = false;
        }
        if (current != (target ? 1 : 0) && !writeGpio(target)) {
            ALOGE("GPIO174 write/readback failed");
            return kIoError;
        }
        return result;
    }

    void reply(int client, Result result) const {
        char line[64];
        const int size = std::snprintf(line, sizeof(line), "%d %d %d %d\n",
                readGpio(), static_cast<int>(mode_),
                static_cast<int>(result), temperature_);
        // 应答很短。短写或断开由客户端作为通信错误处理。
        (void)send(client, line, static_cast<size_t>(size), MSG_NOSIGNAL);
    }

private:
    void setTimedMode(Mode mode, Clock::time_point now) {
        mode_ = mode;
        deadline_ = now + std::chrono::seconds(10);
    }

    Mode mode_ = kAuto;
    int temperature_ = -1;
    Clock::time_point deadline_{};
};

}  // namespace

int main() {
    const int server = android_get_control_socket("h16_fand");
    if (server < 0 || listen(server, 4) < 0) {
        ALOGE("Failed to listen on init socket: %s", std::strerror(errno));
        return EXIT_FAILURE;
    }
    FanControl fan;
    fan.update('?');
    for (;;) {
        pollfd event{server, POLLIN, 0};
        const int ready = poll(&event, 1, 1000);
        if (ready < 0) {
            if (errno == EINTR) continue;
            return EXIT_FAILURE;
        }
        if (ready == 0) {
            fan.update('?');
            continue;
        }
        if ((event.revents & POLLIN) == 0) return EXIT_FAILURE;
        const int client = accept4(server, nullptr, nullptr, SOCK_CLOEXEC);
        if (client < 0) continue;

        // 套接字文件权限和 SELinux 之外，再校验对端 UID。
        ucred peer{};
        socklen_t size = sizeof(peer);
        if (getsockopt(client, SOL_SOCKET, SO_PEERCRED, &peer, &size) == 0 &&
                peer.uid == kSystemUid) {
            const timeval limit{0, 250000};
            setsockopt(client, SOL_SOCKET, SO_RCVTIMEO, &limit, sizeof(limit));
            setsockopt(client, SOL_SOCKET, SO_SNDTIMEO, &limit, sizeof(limit));
            char command;
            if (recv(client, &command, 1, 0) == 1) {
                fan.reply(client, fan.update(command));
            }
        }
        close(client);
        // 即使客户端中途断开，也让定时和温控状态继续推进。
        fan.update('?');
    }
}
```

在 `frameworks/base/Android.mk` 的 Framework AIDL 列表中加入 `core/java/android/os/IH16FanService.aidl`，沿用相邻条目的 Make 续行格式。

```xml
<permission android:name="android.permission.READ_H16_FAN"

```

新建 `frameworks/base/services/core/java/com/android/server/H16FanService.java`。

```java
package com.android.server;

import android.content.Context;
import android.net.LocalSocket;
import android.net.LocalSocketAddress;
import android.os.Binder;
import android.os.Build;
import android.os.IH16FanService;
import android.os.Process;
import android.os.ResultReceiver;
import android.os.ShellCallback;
import android.os.ShellCommand;

import com.android.internal.util.DumpUtils;

import java.io.BufferedReader;
import java.io.FileDescriptor;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.PrintWriter;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;

public final class H16FanService extends SystemService {
    private static final String READ = "android.permission.READ_H16_FAN";
    private static final String CONTROL = "android.permission.CONTROL_H16_FAN";
    private static final char[] MODE_COMMAND = {'A', '1', '0'};

    public H16FanService(Context context) {
        super(context);
    }

    @Override
    public void onStart() {
        publishBinderService("h16_fan", new BinderService());
    }

    private int[] request(char command) {
        // 每次独立连接，避免持有 system_server 全局锁。
        try (LocalSocket socket = new LocalSocket()) {
            socket.connect(new LocalSocketAddress(
                    "h16_fand", LocalSocketAddress.Namespace.RESERVED));
            // Android 8.1 在 connect 后才有底层文件描述符。
            socket.setSoTimeout(500);
            socket.getOutputStream().write(command);
            BufferedReader reader = new BufferedReader(
                    new InputStreamReader(socket.getInputStream(), StandardCharsets.US_ASCII));
            String response = reader.readLine();
            if (response == null) return new int[] {-1, -1, 2, -1};
            String[] fields = response.trim().split(" +");
            if (fields.length != 4) return new int[] {-1, -1, 2, -1};
            int[] result = new int[4];
            for (int i = 0; i < fields.length; ++i) {
                result[i] = Integer.parseInt(fields[i]);
            }
            return result;
        } catch (IOException | NumberFormatException failure) {
            return new int[] {-1, -1, 2, -1};
        }
    }

    private final class BinderService extends IH16FanService.Stub {
        @Override
        public int[] getState() {
            getContext().enforceCallingPermission(READ, "Reading H16 fan state");
            return request('?');
        }

        @Override
        public int[] setMode(int mode) {
            getContext().enforceCallingPermission(CONTROL, "Changing H16 fan mode");
            if (mode < 0 || mode >= MODE_COMMAND.length) {
                throw new IllegalArgumentException("Invalid H16 fan mode");
            }
            return request(MODE_COMMAND[mode]);
        }

        @Override
        protected void dump(FileDescriptor fd, PrintWriter out, String[] args) {
            if (!DumpUtils.checkDumpPermission(getContext(), "h16_fan", out)) return;
            out.println("gpio, mode, result, temperature(mC)");
            out.println(Arrays.toString(request('?')));
        }

        @Override
        public void onShellCommand(FileDescriptor in, FileDescriptor out, FileDescriptor err,
                String[] args, ShellCallback callback, ResultReceiver receiver) {
            // 调试接口仅开放给 userdebug/eng 上的 root。普通客户端走签名权限。
            if (!Build.IS_DEBUGGABLE || Binder.getCallingUid() != Process.ROOT_UID) {
                throw new SecurityException("H16 fan shell command requires debug root");
            }
            new ShellCommand() {
                @Override
                public int onCommand(String arg) {
                    if (arg == null) return handleDefaultCommands(null);
                    char command;
                    switch (arg) {
                        case "status": command = '?'; break;
                        case "on": command = '1'; break;
                        case "off": command = '0'; break;
                        case "auto": command = 'A'; break;
                        default: return handleDefaultCommands(arg);
                    }
                    int[] result = request(command);
                    getOutPrintWriter().println(Arrays.toString(result));
                    return result[2] == 0 ? 0 : 1;
                }

                @Override
                public void onHelp() {
                    getOutPrintWriter().println("cmd h16_fan status|on|off|auto");
                    getOutPrintWriter().println("Response: [gpio, mode, result, temperature(mC)]");
                }
            }.exec(this, in, out, err, args, callback, receiver);
        }
    }
}
```

在 `frameworks/base/services/java/com/android/server/SystemServer.java` 的 `startOtherServices()` 中加入这段。文件位于同一个 Java package，通常已有 `SystemProperties` import。

```java
// 此移植机 getprop ro.product.device 返回 arowana-rc。
if ("arowana-rc".equals(SystemProperties.get("ro.product.device"))) {
    mSystemServiceManager.startService(H16FanService.class);
}
```

`onStart()` 只注册 Binder。后端刚开机尚未监听时，查询可先返回未知并让客户端稍后重试。此示例每次处理一条很短的本地事务。若将接口开放给更多客户端，给 Binder 后端调用加有界工作队列、明确的超时和并发上限，避免后端拥塞时占满 Binder 线程。

## 5. 接入 SELinux

示例使用系统分区上的核心服务，以下片段加入 `system/sepolicy/private/`。另一个部署方式是把设备策略纳入产品专用策略生成流程，确保与当前 ramdisk 合并规则一致。

新建 `system/sepolicy/private/h16_fand.te`。

```text
type h16_fand, domain, coredomain;
type h16_fand_exec, exec_type, file_type;
type h16_fand_socket, file_type;
type h16_fan_service, service_manager_type;
type sysfs_h16_fan_value, fs_type, sysfs_type;

init_daemon_domain(h16_fand)
allow h16_fand self:unix_stream_socket create_stream_socket_perms;
allow h16_fand sysfs_h16_fan_value:file rw_file_perms;
allow h16_fand sysfs:file r_file_perms;
allow h16_fand sysfs_thermal:file r_file_perms;

unix_socket_connect(system_server, h16_fand, h16_fand)
add_service(system_server, h16_fan_service)
userdebug_or_eng(`
    allow su h16_fan_service:service_manager find;
    allow shell h16_fan_service:service_manager find;
')
```

将下面两行加入 `system/sepolicy/private/file_contexts`。

```text
/system/bin/h16_fand     u:object_r:h16_fand_exec:s0
/dev/socket/h16_fand     u:object_r:h16_fand_socket:s0
```

将下面一行加入 `system/sepolicy/private/genfs_contexts`。sysfs 的 genfs 路径去掉挂载点 `/sys`，使用实机解析后的路径。

```text
genfscon sysfs /devices/gpiochip0/gpio/gpio174/value u:object_r:sysfs_h16_fan_value:s0
```

让 init 在 boot 时仍能初始化和设置该 value 节点。在同一策略文件中追加以下规则。

```text
allow init sysfs_h16_fan_value:file { getattr open write setattr };
```

将下面一行加入 `system/sepolicy/private/service_contexts`。

```text
h16_fan    u:object_r:h16_fan_service:s0
```

签名客户端还需要由其 SELinux 域查找服务。确定实际测试应用的域后，添加针对该域的 `service_manager find`；Binder 调用权限按现有平台应用策略验证。读温度的示例与当前 ROM 的 `sysfs` 标签一致，仅赋读取权限。若要进一步缩小范围，先核对各温区的实际路径，再为确认的温度节点使用专用标签。合入前实际编译策略并检查拒绝，尤其注意当前 ramdisk 对 service contexts 和文件标签使用哪套文件。

## 7. 实机验证

```bash
# 指定这台 H16，并使用同一个 SDK adb 避免客户端版本冲突。
ADB_BIN="$HOME/Android/Sdk/platform-tools/adb"
ADB_SERIAL='e2d72141'

# 核对开机完成、进程启动、Binder 服务存在，顺便确认旧风扇服务状态。
"$ADB_BIN" -s "$ADB_SERIAL" shell 'getprop sys.boot_completed; getprop init.svc.h16_fand; getprop init.svc.fan; service list' |
  rg '^(1|running|stopped)$|h16_fan'

# 查询 Binder 服务。输出为 [GPIO, 模式, 结果, 温度毫摄氏度]。
"$ADB_BIN" -s "$ADB_SERIAL" shell dumpsys h16_fan

# 查询物理控制节点，用其值与 Binder 返回的第一列对照。
"$ADB_BIN" -s "$ADB_SERIAL" shell 'su 0 cat /sys/class/gpio/gpio174/value'

# 调试版 root 短时打开风扇。在这 10 秒内听声音或观察气流。
"$ADB_BIN" -s "$ADB_SERIAL" shell 'su 0 cmd h16_fan on'
"$ADB_BIN" -s "$ADB_SERIAL" shell 'su 0 cmd h16_fan status'

# 手动回到自动温控，或等待 10 秒后再次查询，第二列预期为 0。
"$ADB_BIN" -s "$ADB_SERIAL" shell 'su 0 cmd h16_fan auto'

# 确认温度低于 60 °C 时再验证短时关闭，高温或读不到温度时预期被拒绝。
"$ADB_BIN" -s "$ADB_SERIAL" shell 'su 0 cmd h16_fan off'

# 查看旧标签、socket 查找或 GPIO 写权限导致的拒绝。
"$ADB_BIN" -s "$ADB_SERIAL" shell 'su 0 dmesg' | rg 'avc:.*(h16_fan|h16_fand)'
```

如果 `service list` 缺少 `h16_fan`，先看 `SystemServer` 日志和 servicemanager 的服务标签。查询返回 `[-1, -1, 2, -1]` 时，查 native 进程状态、`/dev/socket/h16_fand` 和 SELinux 连接拒绝。能查询但写入失败时，对照 GPIO 的属主、权限和标签，确认新进程用户为 `system`。温控验证先用模拟温度值覆盖阈值、传感器失败和超时，再在实机受控复核。
